def sha(data):
    return hashlib.sha256(data).hexdigest()

def read(path):
    return json.loads(path.read_text(encoding='utf-8'))

def save(name, value):
    (O / name).write_text(json.dumps(value, indent=2) + '\n', encoding='utf-8')

def git(repo, *args):
    return subprocess.run(['git', '--no-optional-locks', '-C', str(repo), *args], capture_output=True, check=True).stdout

def check_source(snapshot):
    repo = Path(snapshot['path'])
    assert git(repo, 'rev-parse', 'HEAD').decode().strip() == snapshot['head']
    assert git(repo, 'status', '--porcelain=v1').decode().strip() == snapshot['status'].strip()
    index = Path(git(repo, 'rev-parse', '--path-format=absolute', '--git-path', 'index').decode().strip())
    assert sha(index.read_bytes()) == snapshot['indexSha256']
    for name, expected in snapshot['files'].items():
        data = (repo / name).read_bytes()
        assert {'bytes': len(data), 'sha256': sha(data)} == expected, name
    return {'files': len(snapshot['files']), 'head': snapshot['head'], 'sourceAndIndexStatusExact': True}

def metadata(path):
    package = path.rsplit('node_modules/', 1)[1]
    version = new[path]['version']
    url = 'https://registry.npmjs.org/' + urllib.parse.quote(package, safe='') + '/' + version
    existing = cached.get((package, version))
    if existing:
        data = (cached_dir / Path(existing['file'].replace('\\', '/'))).read_bytes()
        assert sha(data) == existing['sha256']
        origin = 'exact preserved earlier independent GET'
    else:
        req = urllib.request.Request(url, headers={'Accept': 'application/json', 'User-Agent': 'NodeBenchBoilerplate-read-only-lock-review'})
        with urllib.request.urlopen(req, timeout=25) as response:
            assert response.status == 200 and response.url == url
            data = response.read(2_000_001)
            assert len(data) <= 2_000_000
        origin = 'new bounded public exact-version GET'
    name = package.replace('@', '').replace('/', '__') + '-' + version + '.json'
    (O / 'registry' / name).write_bytes(data)
    m = json.loads(data)
    assert m['name'] == package and m['version'] == version
    lock = new[path]
    assert lock['integrity'] == m['dist']['integrity'], path
    assert lock['resolved'] == m['dist']['tarball'], path
    compared = []
    for key in ['os', 'cpu', 'engines', 'dependencies', 'optionalDependencies']:
        assert lock.get(key, {}) == m.get(key, {}), (path, key)
        compared.append(key)
    # npm 10 lock rows do not necessarily serialize registry libc; retain it,
    # rather than falsely claiming identical package/registry object schemas.
    return {'path': path, 'package': package, 'version': version, 'url': url, 'origin': origin, 'file': 'registry/' + name, 'bytes': len(data), 'sha256': sha(data), 'integrityAndTarballExact': True, 'metadataKeysCompared': compared, 'registryLibc': m.get('libc'), 'lockLibc': lock.get('libc')}

def resolve(parent, child):
    # Resolve only package-record addresses, walking actual node_modules ancestors.
    candidate = parent + '/node_modules/' + child
    if candidate in new:
        return candidate
    current = parent
    while '/node_modules/' in current:
        current = current.rsplit('/node_modules/', 1)[0]
        candidate = current + '/node_modules/' + child
        if candidate in new:
            return candidate
    candidate = 'node_modules/' + child
    assert candidate in new, (parent, child)
    return candidate
